Sharp 製測距センサ **GP2Y0D413K0F** のデータシートに基づいた OpenSCAD モデルです。

### モデリング仕様
- **単位**: mm
- **原点**: センサ本体（レンズケース部: 幅 29.45 × 奥行 13.4 × 高さ 13.05 mm）の中心 `(0, 0, 0)`
  - **X軸**: 左右（投光側が $-X$、受光側が $+X$）
  - **Y軸**: 前後（センサ正面・レンズ照射方向が $+Y$、背面が $-Y$）
  - **Z軸**: 上下（コネクタ方向が $-Z$）
- **主要寸法**:
  - 外形: 幅 29.45 mm、全高 18.9 mm (本体高 13.05 mm + コネクタ部 5.85 mm)
  - 投光レンズ中心: 左端から 4.5 mm ($X = -10.225$)
  - 受光レンズ中心: 投光中心から 19.7 mm ($X = +9.475$)
  - レンズ材質: 可視光カットアクリル樹脂（半透明ダークパープル）
  - ケース材質: 導電性ABS（マットブラック）

```openscad
// Sharp Distance Measuring Sensor GP2Y0D413K0F
// Unit: mm, Origin: Center of the main sensor body

$fn = 48;

// --- Dimensions from Datasheet ---
body_w = 29.45;     // Main case width
body_h = 13.05;     // Main case height
total_h = 18.9;     // Total height including connector

lens_case_d = 7.1;  // Rear block depth
snout_base_d = 4.3; // Middle projection depth (6.3 - 2.0)
lens_rim_d = 2.0;   // Front lens rim depth
total_d = lens_case_d + snout_base_d + lens_rim_d; // 13.4 mm

// Snout heights
snout_h = 8.4;
rim_h = 7.2;

// X coordinates derived from datasheet
// Left margin: 0.75 mm, Emitter snout width: 7.5 mm -> Center at -14.725 + 4.5 = -10.225 mm
emitter_x = -body_w/2 + 4.5; // -10.225
detector_x = emitter_x + 19.7; // +9.475

// Projection widths and positions
emitter_snout_w = 7.5;
emitter_snout_x = -body_w/2 + 0.75 + emitter_snout_w/2; // -10.225

detector_snout_w = 16.3;
detector_snout_x = body_w/2 - 0.75 - detector_snout_w/2; // +5.825

// Connector dimensions
conn_w = 10.1;
conn_h = total_h - body_h; // 5.85 mm
conn_d = 4.6;
pwb_t = 1.2;

// --- Color Palette ---
c_case  = [0.20, 0.20, 0.22];        // Carbonic ABS (Dark gray / Black)
c_lens  = [0.25, 0.05, 0.12, 0.85];   // Visible light cut-off acrylic (Dark red-purple)
c_conn  = [0.90, 0.90, 0.88];        // Connector housing (Natural white/ivory)
c_pin   = [0.85, 0.75, 0.35];        // Metal terminal pins (Gold/Brass)
c_pwb   = [0.35, 0.25, 0.15];        // Paper phenol board (Brown)

// --- Assembly ---
module gp2y0d413k0f() {
    main_case();
    lenses();
    connector();
}

// 1. Main Case Body (Conductive ABS)
module main_case() {
    color(c_case) {
        difference() {
            union() {
                // Rear main body
                translate([0, -total_d/2 + lens_case_d/2, 0])
                    cube([body_w, lens_case_d, body_h], center=true);

                // Intermediate projections (Height: 8.4mm)
                y_snout = -total_d/2 + lens_case_d + snout_base_d/2;
                
                // Emitter side base
                translate([emitter_snout_x, y_snout, 0])
                    cube([emitter_snout_w, snout_base_d, snout_h], center=true);

                // Detector side base
                translate([detector_snout_x, y_snout, 0])
                    cube([detector_snout_w, snout_base_d, snout_h], center=true);

                // Front lens rims (Height: 7.2mm)
                y_rim = total_d/2 - lens_rim_d/2;
                
                // Emitter cylindrical barrel
                translate([emitter_x, y_rim, 0])
                    rotate([-90, 0, 0])
                        cylinder(d=rim_h, h=lens_rim_d, center=true);

                // Detector rectangular barrel with rounded corners
                translate([detector_snout_x, y_rim, 0])
                    hull() {
                        translate([-(detector_snout_w - rim_h)/2, 0, 0])
                            rotate([-90, 0, 0]) cylinder(d=rim_h, h=lens_rim_d, center=true);
                        translate([(detector_snout_w - rim_h)/2, 0, 0])
                            rotate([-90, 0, 0]) cylinder(d=rim_h, h=lens_rim_d, center=true);
                    }
            }

            // Lens apertures (Cuts for optical lens insertion)
            y_front = total_d/2;
            
            // Emitter lens aperture
            translate([emitter_x, y_front - lens_rim_d/2 + 0.1, 0])
                rotate([-90, 0, 0])
                    cylinder(d=5.5, h=lens_rim_d + 0.2, center=true);

            // Detector lens aperture (Slit window)
            translate([detector_snout_x, y_front - lens_rim_d/2 + 0.1, 0])
                hull() {
                    translate([-(detector_snout_w - rim_h - 1)/2, 0, 0])
                        rotate([-90, 0, 0]) cylinder(d=5.2, h=lens_rim_d + 0.2, center=true);
                    translate([(detector_snout_w - rim_h - 1)/2, 0, 0])
                        rotate([-90, 0, 0]) cylinder(d=5.2, h=lens_rim_d + 0.2, center=true);
                }
        }
    }
}

// 2. Optical Lenses (Acrylic Resin)
module lenses() {
    color(c_lens) {
        y_lens = total_d/2 - lens_rim_d/2;

        // Emitter convex lens
        translate([emitter_x, y_lens, 0]) {
            rotate([-90, 0, 0])
                cylinder(d=5.4, h=lens_rim_d - 0.2, center=true);
            // Convex bulge
            translate([0, lens_rim_d/2 - 0.3, 0])
                scale([1, 0.45, 1])
                    sphere(d=5.0);
        }

        // Detector lens (Aspheric / cylindrical receiver lens)
        translate([detector_snout_x, y_lens, 0]) {
            hull() {
                translate([-(detector_snout_w - rim_h - 1.2)/2, 0, 0])
                    rotate([-90, 0, 0]) cylinder(d=5.1, h=lens_rim_d - 0.2, center=true);
                translate([(detector_snout_w - rim_h - 1.2)/2, 0, 0])
                    rotate([-90, 0, 0]) cylinder(d=5.1, h=lens_rim_d - 0.2, center=true);
            }
        }
        // Detector focused lens bulge (at detector_x: +9.475)
        translate([detector_x, y_lens + lens_rim_d/2 - 0.3, 0])
            scale([1, 0.45, 1])
                sphere(d=5.0);
    }
}

// 3. Connector & PWB
module connector() {
    // PWB Board
    color(c_pwb) {
        translate([0, -total_d/2 + 3.3 + pwb_t/2, -body_h/2 - conn_h/2])
            cube([conn_w - 0.4, pwb_t, conn_h], center=true);
    }

    // Connector Housing (JCTC 12001W90-3P-HF)
    color(c_conn) {
        difference() {
            // Connector body
            translate([0, -total_d/2 + 3.3 + pwb_t + conn_d/2, -body_h/2 - conn_h/2])
                cube([conn_w, conn_d, conn_h], center=true);

            // Inner mating slot
            translate([0, -total_d/2 + 3.3 + pwb_t + conn_d/2 + 0.6, -body_h/2 - conn_h/2 - 0.6])
                cube([conn_w - 1.6, conn_d - 0.8, conn_h], center=true);
        }
    }

    // Terminal Pins (3 Poles: ①Vo, ②GND, ③Vcc)
    color(c_pin) {
        for (i = [-1, 0, 1]) {
            // Pin pitch: 1.5 mm
            translate([i * 1.5, -total_d/2 + 3.3 + pwb_t + conn_d/2 + 0.5, -body_h/2 - conn_h/2])
                cube([0.4, 0.4, conn_h - 1.5], center=true);
            // Pins extending downward
            translate([i * 1.5, -total_d/2 + 3.3 + pwb_t/2, -body_h/2 - conn_h - 0.5])
                cube([0.4, 0.4, 1.2], center=true);
        }
    }
}

// Render model
gp2y0d413k0f();
```
