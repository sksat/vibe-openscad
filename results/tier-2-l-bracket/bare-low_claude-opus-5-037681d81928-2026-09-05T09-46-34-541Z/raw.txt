```openscad
// L-bracket with countersunk holes
$fn = 64;

W  = 50;   // 幅
D  = 40;   // 水平面の奥行き
H  = 40;   // 垂直面の高さ
T  = 3;    // 板厚

HOLE_D = 4.5;   // 貫通穴
CS_D   = 8;     // 皿座面径
CS_H   = 2;     // 皿座面深さ
EDGE   = 10;    // 縁からの距離

module countersunk_hole() {
    // 原点が外側表面、+Z 方向が材料内部
    translate([0, 0, -1])
        cylinder(d = HOLE_D, h = T + 2);
    translate([0, 0, -0.01])
        cylinder(d1 = CS_D, d2 = HOLE_D, h = CS_H);
    translate([0, 0, -1])
        cylinder(d = CS_D, h = 1.01);
}

difference() {
    union() {
        // 水平フランジ(+Y 方向、内側面が z=0)
        translate([-W/2, 0, -T]) cube([W, D, T]);
        // 垂直フランジ(+Z 方向、内側面が y=0)
        translate([-W/2, -T, 0]) cube([W, T, H]);
    }

    // 水平面の皿穴(外側 = 下面 z=-T)
    for (x = [-(W/2 - EDGE), (W/2 - EDGE)])
        translate([x, D/2, -T])
            countersunk_hole();

    // 垂直面の皿穴(外側 = 背面 y=-T)
    for (x = [-(W/2 - EDGE), (W/2 - EDGE)])
        translate([x, -T, H/2])
            rotate([-90, 0, 0])
                countersunk_hole();
}
```
