```openscad
// ============================================================
//  L-bracket (L字金具) with countersunk holes
//  内側コーナー = 原点 / 水平面 = +Y方向 / 垂直面 = +Z方向
// ============================================================

$fn = 64;

// ---- 主要寸法 ----
W    = 50;   // 幅 (X)
D    = 40;   // 水平面の奥行き (Y)
H    = 40;   // 垂直面の高さ (Z)
T    = 3;    // 板厚

// ---- 皿穴 (M4) ----
HOLE_D  = 4.5;   // 貫通穴径
CSK_D   = 8;     // 皿座面径
CSK_H   = 2;     // 皿座面深さ
EDGE    = 10;    // 面の縁からの距離
EPS     = 0.01;

HX = W/2 - EDGE;   // ±15mm : 左右対称位置

// ------------------------------------------------------------
// 皿穴カッター
//   ローカル座標: 外側表面 = z=0、材料は +Z 方向へ厚さ t
// ------------------------------------------------------------
module csk_cutter(t) {
    // 貫通穴
    translate([0, 0, -1])
        cylinder(d = HOLE_D, h = t + 2);
    // 皿座面 (テーパ)
    translate([0, 0, -EPS])
        cylinder(d1 = CSK_D, d2 = HOLE_D, h = CSK_H + EPS);
    // 表面より外側は完全に開口させる
    translate([0, 0, -1])
        cylinder(d = CSK_D, h = 1 + EPS);
}

// ------------------------------------------------------------
// 本体
// ------------------------------------------------------------
module l_bracket() {
    difference() {
        union() {
            // 水平フランジ (内側上面が z=0)
            translate([-W/2, 0, -T])
                cube([W, D, T]);
            // 垂直フランジ (内側前面が y=0) + コーナー部
            translate([-W/2, -T, -T])
                cube([W, T, H + T]);
        }

        // --- 水平面の皿穴 (皿座面は下面 = 外側) ---
        for (x = [-HX, HX])
            translate([x, D/2, -T])
                csk_cutter(T);

        // --- 垂直面の皿穴 (皿座面は背面 = 外側) ---
        for (x = [-HX, HX])
            translate([x, -T, H/2])
                rotate([-90, 0, 0])
                    csk_cutter(T);
    }
}

l_bracket();
```
