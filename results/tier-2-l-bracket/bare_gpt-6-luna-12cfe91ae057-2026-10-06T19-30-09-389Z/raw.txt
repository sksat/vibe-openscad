```openscad
$fn = 64;

width = 50;
depth = 40;
height = 40;
thickness = 3;

hole_diameter = 4.5;
countersink_diameter = 8;
countersink_depth = 2;

module bracket() {
    union() {
        // Horizontal flange: extends in +Y; inside face is at Z=0.
        translate([-width/2, 0, -thickness])
            cube([width, depth, thickness]);

        // Vertical flange: extends in +Z; inside face is at Y=0.
        translate([-width/2, -thickness, 0])
            cube([width, thickness, height]);

        // Material at the outside corner joins the two flanges.
        translate([-width/2, -thickness, -thickness])
            cube([width, thickness, thickness]);
    }
}

difference() {
    bracket();

    // Horizontal flange holes, countersinks facing outward (-Z).
    for (y = [10, depth - 10]) {
        translate([0, y, -thickness - 0.1])
            cylinder(h = thickness + 0.2, d = hole_diameter);

        translate([0, y, -thickness])
            cylinder(
                h = countersink_depth,
                d1 = countersink_diameter,
                d2 = hole_diameter
            );
    }

    // Vertical flange holes, countersinks facing outward (-Y).
    for (z = [10, height - 10]) {
        translate([0, -thickness - 0.1, z])
            rotate([-90, 0, 0])
                cylinder(h = thickness + 0.2, d = hole_diameter);

        translate([0, -thickness, z])
            rotate([-90, 0, 0])
                cylinder(
                    h = countersink_depth,
                    d1 = countersink_diameter,
                    d2 = hole_diameter
                );
    }
}
```
