```openscad
// ============================================================
// L-Bracket (L字金具)
//   水平フランジ : 50 (X) x 40 (Y) x 3 (t)
//   垂直フランジ : 50 (X) x 3 (t) x 40 (Z)
//   内側コーナーが原点、水平面は +Y、垂直面は +Z へ伸びる
//   皿穴 M4 : 通し穴 Φ4.5 / 皿座 Φ8 x 深さ2 (外側向き)
// ============================================================

$fn = 64;

// ---- パラメータ ----------------------------------------
W        = 50;   // 幅 (X方向)
D        = 40;   // 水平フランジ奥行き (Y方向)
H        = 40;   // 垂直フランジ高さ (Z方向)
T        =  3;   // 板厚

HOLE_D   = 4.5;  // 貫通穴径
CS_D     = 8;    // 皿座面の径
CS_DEPTH = 2;    // 皿座面の深さ

EDGE     = 10;   // 面の縁からの距離
PITCH    = 30;   // 穴の左右ピッチ（中心対称）

EPS      = 0.01; // z-fighting 回避用

// ---- 皿穴モジュール ------------------------------------
// 原点を「皿座面のある外表面」に置き、-Z 方向へ材料が続く前提。
// 皿座面は +Z 側（外側）に開く。
module countersunk_hole(thickness = T) {
    // 貫通穴
    translate([0, 0, -thickness - EPS])
        cylinder(h = thickness + 2 * EPS, d = HOLE_D);

    // 皿座（テーパ部）: 外表面で Φ8 → 深さ2 で Φ4.5
    translate([0, 0, -CS_DEPTH])
        cylinder(h = CS_DEPTH, d1 = HOLE_D, d2 = CS_D);

    // 表面より上へ少し延長（開口を確実に開ける）
    cylinder(h = EPS * 2, d = CS_D);
}

// ---- 本体 ----------------------------------------------
module l_bracket() {
    difference() {
        union() {
            // 水平フランジ : Z = 0 .. -T の位置ではなく、
            // 内側コーナーを原点とするため Z = -T .. 0 に配置
            translate([-W/2, 0, -T]) cube([W, D, T]);
            // 垂直フランジ : Y = -T .. 0 に配置
            translate([-W/2, -T, -T]) cube([W, T, H + T]);
        }

        // --- 水平フランジの皿穴 (外側 = 下面 Z = -T) ---
        for (x = [-PITCH/2, PITCH/2])
            translate([x, D - EDGE, -T])
                rotate([180, 0, 0])          // 皿座を下向き(外側)へ
                    countersunk_hole(T);

        // --- 垂直フランジの皿穴 (外側 = 背面 Y = -T) ---
        for (x = [-PITCH/2, PITCH/2])
            translate([x, -T, H - EDGE])
                rotate([90, 0, 0])           // 皿座を -Y (外側)へ
                    countersunk_hole(T);
    }
}

l_bracket();
```
