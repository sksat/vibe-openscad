```openscad
$fn = 96;

width = 50;
depth = 40;
height = 40;
thickness = 3;

hole_d = 4.5;
countersink_d = 8;
countersink_depth = 2;
edge_offset = 10;
eps = 0.01;

// 内側コーナーを Y=0, Z=0 に置く。
difference() {
    union() {
        // 水平フランジ：内側コーナーから +Y 方向へ 40 mm
        translate([-width/2, -thickness, -thickness])
            cube([width, depth + thickness, thickness]);

        // 垂直フランジ：内側コーナーから +Z 方向へ 40 mm
        translate([-width/2, -thickness, -thickness])
            cube([width, thickness, height + thickness]);
    }

    for (x = [-width/2 + edge_offset, width/2 - edge_offset]) {
        // 水平面：外側（下面）に皿座面
        translate([x, depth/2, -thickness - eps])
            cylinder(h = thickness + 2*eps, d = hole_d);
        translate([x, depth/2, -thickness])
            cylinder(h = countersink_depth,
                     d1 = countersink_d, d2 = hole_d);

        // 垂直面：外側（背面）に皿座面
        translate([x, -thickness - eps, height/2])
            rotate([-90, 0, 0])
                cylinder(h = thickness + 2*eps, d = hole_d);
        translate([x, -thickness, height/2])
            rotate([-90, 0, 0])
                cylinder(h = countersink_depth,
                         d1 = countersink_d, d2 = hole_d);
    }
}
```
