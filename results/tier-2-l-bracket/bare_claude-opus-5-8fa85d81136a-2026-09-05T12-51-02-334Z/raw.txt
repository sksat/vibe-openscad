```openscad
// =====================================================
//  L-bracket (L 字金具)  M4 皿穴 x 4
// =====================================================
$fn = 96;

W        = 50;   // 幅 (X)
D        = 40;   // 水平面の奥行き (Y)
H        = 40;   // 垂直面の高さ  (Z)
T        = 3;    // 板厚
HOLE_D   = 4.5;  // 貫通穴径
CSK_D    = 8;    // 皿座面径
CSK_DP   = 2;    // 皿座面深さ
EDGE     = 10;   // 面の縁からの距離

XOFF = W/2 - EDGE;   // ±15 : 左右の穴位置
YPOS = D/2;          // 水平面 中心線
ZPOS = H/2;          // 垂直面 中心線

// -----------------------------------------------------
// 皿穴 : +Z 方向へ貫通、z=0 側(外側)に皿座面
// -----------------------------------------------------
module countersunk_hole(t = T) {
    // 貫通穴
    translate([0, 0, -1])
        cylinder(d = HOLE_D, h = t + 2);
    // 皿座面(円錐)
    cylinder(d1 = CSK_D, d2 = HOLE_D, h = CSK_DP);
    // 外側の逃げ
    translate([0, 0, -1])
        cylinder(d = CSK_D, h = 1);
}

// -----------------------------------------------------
// 本体
// -----------------------------------------------------
difference() {

    union() {
        // 水平フランジ : 内面 z=0 、外面 z=-T 、+Y 方向へ
        translate([-W/2, -T, -T])
            cube([W, D + T, T]);

        // 垂直フランジ : 内面 y=0 、外面 y=-T 、+Z 方向へ
        translate([-W/2, -T, -T])
            cube([W, T, H + T]);
    }

    // --- 水平面の皿穴(皿座面は下面 z=-T)---
    for (x = [-XOFF, XOFF])
        translate([x, YPOS, -T])
            countersunk_hole(T);

    // --- 垂直面の皿穴(皿座面は手前面 y=-T)---
    for (x = [-XOFF, XOFF])
        translate([x, -T, ZPOS])
            rotate([-90, 0, 0])
                countersunk_hole(T);
}
```
