```openscad
// 単位: mm
// 内側コーナー: 原点
// 幅方向: X（左右対称）、水平面: +Y、垂直面: +Z
$fn = 96;

width       = 50;
depth       = 40;
height      = 40;
thickness   = 3;

hole_d      = 4.5;
head_d      = 8;
sink_depth  = 2;
edge_offset = 10;
eps         = 0.01;

// ローカル +Z が板の内側を向く皿穴。
// z = 0 が外側表面。
module countersunk_hole() {
    union() {
        translate([0, 0, -eps])
            cylinder(h = thickness + 2 * eps, d = hole_d);

        cylinder(h = sink_depth, d1 = head_d, d2 = hole_d);

        translate([0, 0, -eps])
            cylinder(h = eps, d = head_d);
    }
}

difference() {
    union() {
        // 水平フランジ: 上面が z = 0
        translate([-width / 2, 0, -thickness])
            cube([width, depth, thickness]);

        // 垂直フランジ: 内側表面が y = 0
        translate([-width / 2, -thickness, 0])
            cube([width, thickness, height]);

        // 両フランジを接続するコーナー部分
        translate([-width / 2, -thickness, -thickness])
            cube([width, thickness, thickness]);
    }

    for (x = [-width / 2 + edge_offset,
               width / 2 - edge_offset]) {
        // 水平面: 皿座面は下面（-Z 側）
        translate([x, depth / 2, -thickness])
            countersunk_hole();

        // 垂直面: 皿座面は背面（-Y 側）
        translate([x, -thickness, height / 2])
            rotate([-90, 0, 0])
                countersunk_hole();
    }
}
```
