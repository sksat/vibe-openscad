```openscad
$fn = 64;

// Dimensions
width = 50;
depth = 40;
height = 40;
thickness = 3;

// Hole specs
through_d = 4.5;
csink_d = 8;
csink_depth = 2;

// Hole positions
x1 = 10;
x2 = width - 10;
y_center = depth / 2;
z_center = height / 2;

module countersunk_hole_z(h=thickness) {
    union() {
        cylinder(h=csink_depth, d=csink_d);
        translate([0, 0, csink_depth])
            cylinder(h=h - csink_depth, d=through_d);
    }
}

module countersunk_hole_y(h=thickness) {
    rotate([-90, 0, 0])
        union() {
            cylinder(h=csink_depth, d=csink_d);
            translate([0, 0, csink_depth])
                cylinder(h=h - csink_depth, d=through_d);
        }
}

difference() {
    union() {
        // Horizontal flange: inner face at z=0, extends +Y, thickness toward -Z
        translate([0, 0, -thickness])
            cube([width, depth, thickness]);

        // Vertical flange: inner face at y=0, extends +Z, thickness toward -Y
        translate([0, -thickness, 0])
            cube([width, thickness, height]);
    }

    // Horizontal flange holes
    // centerline of the face, 10mm from front/back edges along Y
    // countersink on outer face (z = -thickness)
    for (yy = [10, depth - 10])
        translate([width / 2, yy, -thickness])
            countersunk_hole_z(thickness);

    // Vertical flange holes
    // centerline of the face, 10mm from bottom/top edges along Z
    // countersink on outer face (y = -thickness)
    for (zz = [10, height - 10])
        translate([width / 2, -thickness, zz])
            countersunk_hole_y(thickness);
}
```
