```openscad
// 3段の階段状ピラミッド
// 各段の高さ (mm)
h = 10;

union() {
    // 1段目（底辺）: 60mm × 60mm × 10mm
    translate([-30, -30, 0])
        cube([60, 60, h]);

    // 2段目: 40mm × 40mm × 10mm（中心を揃えて1段目の上に載せる）
    translate([-20, -20, h])
        cube([40, 40, h]);

    // 3段目（最上段）: 20mm × 20mm × 10mm（中心を揃えて2段目の上に載せる）
    translate([-10, -10, 2 * h])
        cube([20, 20, h]);
}
```
