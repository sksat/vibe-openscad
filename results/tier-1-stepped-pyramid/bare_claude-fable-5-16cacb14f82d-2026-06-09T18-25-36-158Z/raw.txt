```openscad
// 3段の階段状ピラミッド
// 各段の中心を揃えて積み上げる

// 1段目: 60mm x 60mm x 高さ10mm
translate([0, 0, 0])
    cube([60, 60, 10], center = false);

// 2段目: 40mm x 40mm x 高さ10mm(中心を揃える)
translate([10, 10, 10])
    cube([40, 40, 10], center = false);

// 3段目: 20mm x 20mm x 高さ10mm(中心を揃える)
translate([20, 20, 20])
    cube([20, 20, 10], center = false);
```
