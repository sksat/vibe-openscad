```openscad
// M8 六角ボルト (ねじ山なし)

// 頭部の円周方向の滑らかさを設定
$fn = 64;

// 頭部: 対辺距離 13mm、高さ 5.3mm
// OpenSCADのcylinder(fn=6)は「半径」を指定するため、対辺距離(13mm)の半分である「内接円半径」を計算
// 内接円半径 = 対辺距離 / 2 = 6.5mm
cylinder(r = 6.5 / cos(180/6), h = 5.3, $fn = 6);

// シャンク: 直径 8mm、長さ 30mm
// z=0の位置から z<0 側へ伸ばす
translate([0, 0, -30])
cylinder(r = 8 / 2, h = 30);
```
