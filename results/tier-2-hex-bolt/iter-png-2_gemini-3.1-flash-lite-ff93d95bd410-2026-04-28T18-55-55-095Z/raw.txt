```openscad
// M8 Hex Bolt
// 寸法：二面幅13mm、頭部高さ5.3mm、シャンク径8mm、長さ30mm
// 配置：頭部底面が z=0、シャンクが z<0 方向

head_af = 13;
head_height = 5.3;
shank_dia = 8;
shank_len = 30;

// cylinder(..., $fn=6) を使用する場合の半径 r = (二面幅 / 2) / cos(180/6)
// つまり r = (head_af / 2) / (sqrt(3)/2) = head_af / sqrt(3)
hex_radius = head_af / sqrt(3);

union() {
    // 頭部: z=0 から z=5.3 まで
    cylinder(r = hex_radius, h = head_height, $fn = 6);
    
    // シャンク: z=0 から z=-30 まで
    translate([0, 0, -shank_len])
        cylinder(d = shank_dia, h = shank_len, $fn = 50);
}
```
