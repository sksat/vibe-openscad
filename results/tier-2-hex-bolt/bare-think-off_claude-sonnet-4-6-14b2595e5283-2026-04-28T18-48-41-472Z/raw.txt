```openscad
// M8 Hex Bolt (threads omitted, cylindrical shank)
// Head bottom face at z=0, shank extends in -Z direction

// Parameters
across_flats = 13;    // Head across-flats distance (mm)
head_height  = 5.3;   // Head height (mm)
shank_dia    = 8;     // Shank diameter (mm)
shank_length = 30;    // Shank length (mm)

// Circumradius of regular hexagon from across-flats distance
hex_r = across_flats / (2 * cos(30));

// Hex head: from z=0 to z=+head_height
module hex_head() {
    linear_extrude(height = head_height) {
        circle(r = hex_r, $fn = 6);
    }
}

// Shank: from z=0 to z=-shank_length
module shank() {
    translate([0, 0, -shank_length]) {
        cylinder(h = shank_length, d = shank_dia, $fn = 64);
    }
}

// Assemble
union() {
    hex_head();
    shank();
}
```
