```openscad
// M8 六角ボルト (ねじ山省略・円柱シャンク)
// 中心軸: +Z, 頭部底面: z=0, シャンク: z<0 側

head_af = 13;    // 頭部 対辺距離 (across-flats) [mm]
head_h  = 5.3;   // 頭部 高さ [mm]
shank_d = 8;     // シャンク 直径 [mm]
shank_l = 30;    // シャンク 長さ [mm]

eps = 0.01;      // 面の共有による描画不具合を避けるための微小重なり

module hex_bolt_m8() {
    union() {
        // 頭部: 六角柱 (底面 z=0 ～ 上面 z=head_h)
        // cylinder の d は外接円直径なので、対辺距離から換算する
        cylinder(h = head_h, d = head_af / cos(30), $fn = 6);

        // シャンク: 円柱 (z = -shank_l ～ z = 0、頭部内へ eps だけ食い込ませる)
        translate([0, 0, -shank_l])
            cylinder(h = shank_l + eps, d = shank_d, $fn = 96);
    }
}

hex_bolt_m8();
```
