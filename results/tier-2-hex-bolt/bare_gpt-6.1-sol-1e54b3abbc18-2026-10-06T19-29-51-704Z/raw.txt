```openscad
head_af = 13;
head_height = 5.3;
shank_diameter = 8;
shank_length = 30;

union() {
    // 頭部：底面 z=0、上面 z=5.3
    rotate([0, 0, 30])
        cylinder(
            h = head_height,
            r = head_af / sqrt(3),
            $fn = 6
        );

    // シャンク：z=-30 から z=0
    translate([0, 0, -shank_length])
        cylinder(
            h = shank_length,
            d = shank_diameter,
            $fn = 96
        );
}
```
