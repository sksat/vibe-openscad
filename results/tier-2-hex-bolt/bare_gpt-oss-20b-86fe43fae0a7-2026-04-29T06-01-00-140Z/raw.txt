```openscad
// M8 六角ボルト（規格に準じた寸法）
module hex_prism() {
    // 対辺距離が 13mm の正六角形の1辺長
    side = 13 / sqrt(3);
    pts = [for (i = [0:5]) [cos(i * 60) * side, sin(i * 60) * side]];
    linear_extrude(height = 5.3)
        polygon(points = pts);
}

module bolt() {
    // 六角ヘッド（底面が z=0）
    translate([0, 0, 5.3 / 2])
        hex_prism();

    // シャンク（z<0 側へ伸びる円柱）
    translate([0, 0, -15])  // 中心を z=-15 に置く
        cylinder(h = 30, r = 4, center = false);
}

bolt();
```
