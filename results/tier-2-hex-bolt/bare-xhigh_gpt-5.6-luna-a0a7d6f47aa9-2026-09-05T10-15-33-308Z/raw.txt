```openscad
af = 13;
head_height = 5.3;
shaft_diameter = 8;
shaft_length = 30;
hex_radius = af / sqrt(3);

union() {
    translate([0, 0, -shaft_length])
        cylinder(
            d = shaft_diameter,
            h = shaft_length,
            $fn = 64
        );

    cylinder(
        r = hex_radius,
        h = head_height,
        $fn = 6
    );
}
```
