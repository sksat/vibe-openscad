```openscad
// M8 六角ボルト(ねじ山省略、円柱シャンク)

// パラメータ
af = 13;    // 頭部 対辺距離 (across-flats) [mm]
hh = 5.3;   // 頭部 高さ [mm]
sd = 8;     // シャンク直径 [mm]
sl = 30;    // シャンク長さ [mm]

$fn = 64;   // 円柱の滑らかさ

module hex_bolt() {
    union() {
        // 頭部: 六角柱(底面 z=0、上面 z=hh)
        // $fn=6 の cylinder は外接円基準なので、対辺距離から外接円直径へ変換
        cylinder(h = hh, d = af / cos(30), $fn = 6);

        // シャンク: z=-sl から z=0 まで
        translate([0, 0, -sl])
            cylinder(h = sl, d = sd);
    }
}

hex_bolt();
```
