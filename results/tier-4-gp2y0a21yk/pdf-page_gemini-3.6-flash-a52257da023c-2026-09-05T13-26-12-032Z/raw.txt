添付いただいた Sharp 製測距センサ **GP2Y0A21YK0F** のデータシート寸法に基づいて作成した OpenSCAD モデルのコードです。

本体中心を原点 (0, 0, 0) とし、外形寸法・レンズ部・取付耳・基板 (PWB)・コネクタ部を忠実に再現しています。

```openscad
// Sharp GP2Y0A21YK0F Distance Measuring Sensor
// Units: mm
// Origin: Center of the main sensor body

$fn = 40;

// --- Dimension Parameters (from datasheet) ---
// Main Body
body_w = 29.5;       // Main housing width
body_h = 13.0;       // Main housing height
body_d = 6.3;        // Main housing depth (excluding lens protrusion)

// Mounting Flanges / Ears
hole_pitch = 29.5;   // Distance between hole centers
ear_r = 3.75;        // Flange outer radius
hole_dia = 3.2;      // Mounting hole diameter
ear_thick = 1.5;     // Flange thickness

// Lenses
lens_pitch = 20.0;   // Distance between emitter and detector lens centers
lens_h = 7.2;        // Height of lens housing frame
lens_protrusion = 2.0; // Protrusion from body front
emitter_w = 7.5;     // Emitter housing width
detector_w = 16.3;   // Detector housing width

// PWB (Printed Wiring Board)
pwb_w = 29.5;
pwb_h = 13.0;
pwb_thick = 1.2;

// Connector
conn_w = 10.1;
conn_h = 5.9;        // Extends below body (total height ~18.9mm)
conn_d = 6.0;

// --- Color Definitions ---
case_color = [0.18, 0.18, 0.18];    // Conductive ABS (Dark Grey/Black)
lens_color = [0.12, 0.08, 0.15, 0.9]; // Dark Acrylic Resin
pwb_color  = [0.55, 0.40, 0.20];    // Paper Phenol (Brown)
conn_color = [0.90, 0.90, 0.85];    // White Plastic Connector
pin_color  = [0.85, 0.75, 0.30];    // Metal Pins

module gp2y0a21yk0f() {
    union() {
        // 1. Main Plastic Body & Bezels
        color(case_color) {
            // Central block
            cube([body_w, body_h, body_d], center=true);

            // Left Mounting Flange (Ear)
            translate([-hole_pitch/2, 0, -body_d/2 + ear_thick/2]) {
                difference() {
                    hull() {
                        cylinder(r=ear_r, h=ear_thick, center=true);
                        translate([ear_r, 0, 0])
                            cube([ear_r*2, ear_r*2, ear_thick], center=true);
                    }
                    cylinder(d=hole_dia, h=ear_thick + 0.2, center=true);
                }
            }

            // Right Mounting Flange (Ear)
            translate([hole_pitch/2, 0, -body_d/2 + ear_thick/2]) {
                difference() {
                    hull() {
                        cylinder(r=ear_r, h=ear_thick, center=true);
                        translate([-ear_r, 0, 0])
                            cube([ear_r*2, ear_r*2, ear_thick], center=true);
                    }
                    cylinder(d=hole_dia, h=ear_thick + 0.2, center=true);
                }
            }

            // Lens Housing Frames (Front side)
            // Left (Emitter) Housing Frame
            translate([-lens_pitch/2 + 0.25, 0, body_d/2 + lens_protrusion/2])
                cube([emitter_w, lens_h, lens_protrusion], center=true);

            // Right (Detector) Housing Frame
            translate([lens_pitch/2 - 0.15, 0, body_d/2 + lens_protrusion/2])
                cube([detector_w, lens_h, lens_protrusion], center=true);
        }

        // 2. Optical Lenses (Inside the housings)
        color(lens_color) {
            // Emitter Lens (Circular Convex)
            translate([-lens_pitch/2, 0, body_d/2 + lens_protrusion - 0.2])
                scale([1, 1, 0.4])
                    sphere(r=2.7);

            // Detector Lens (Elliptical Convex)
            translate([lens_pitch/2, 0, body_d/2 + lens_protrusion - 0.2])
                scale([2.2, 1, 0.4])
                    sphere(r=2.7);
        }

        // 3. PWB (Back of main body)
        color(pwb_color) {
            translate([0, 0, -body_d/2 - pwb_thick/2])
                cube([pwb_w, pwb_h, pwb_thick], center=true);
        }

        // 4. Connector (Bottom side)
        color(conn_color) {
            translate([0, -body_h/2 - conn_h/2 + 1.0, -body_d/2 + conn_d/2 - pwb_thick]) {
                difference() {
                    cube([conn_w, conn_h, conn_d], center=true);
                    // Inner cavity for header
                    translate([0, -1, 0])
                        cube([conn_w - 1.6, conn_h, conn_d - 1.6], center=true);
                }
            }
        }

        // Connector Pins (3-pin)
        color(pin_color) {
            for (i = [-1:1]) {
                translate([i * 2.5, -body_h/2 - 2.0, -body_d/2 + conn_d/2 - pwb_thick])
                    cube([0.6, 3.5, 0.6], center=true);
            }
        }
    }
}

// Render the module
gp2y0a21yk0f();
```
