    color("dimgray")
    for (x = [-10, 10]) {
        translate([x, 8.4/2 + 2/2, 7.2/2 + 2 + 7.2/2])
        cylinder(d=7, h=2, center=true);

    // 下部コネクタ部
    color("black")
