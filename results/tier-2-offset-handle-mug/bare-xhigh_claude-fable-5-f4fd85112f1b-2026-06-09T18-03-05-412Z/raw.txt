// ============================================

$fn = 96;

// ---- 本体パラメータ ----

inner_d  = 70;   // 内径(肉厚 5mm)
mug_h    = 90;   // 高さ
bottom_t = 6;    // 底面の厚み

// ---- 取手パラメータ ----
grip_h   = 30;   // 取手内側空間の高さ
grip_w   = 25;   // 取手内側空間の幅(本体表面から)
bar_t    = 8;    // 取手の断面の太さ(X-Z 方向)
handle_y = 12;   // 取手の奥行き(Y 方向の厚み)

embed_x  = 34;   // 取手の根元 X 位置(壁内に食い込ませて確実に union)
handle_z = mug_h / 2;  // 高さ方向の中央に配置

// 内側 D 形状: 左端 x=34 → 開口部は本体表面(x=40)から x=65 まで = 幅 25
hi_h = grip_h;                      // 30
hi_w = (40 - embed_x) + grip_w;     // 31
// 外側 D 形状: 全周に bar_t を足す
ho_h = hi_h + 2 * bar_t;            // 46
ho_w = hi_w + bar_t;                // 39

// ---- D 字形 2D 形状(左辺が直線、右側が半円)----
module d2(w, h) {
    union() {
        translate([0, -h/2]) square([w - h/2, h]);
        translate([w - h/2, 0]) circle(d = h);
    }
}

// ---- 取手(+X 方向、X-Z 平面の D 字リングを Y 方向に押し出し)----
module handle() {
    translate([embed_x, 0, handle_z])

            linear_extrude(height = handle_y, center = true)
                difference() {
                    d2(ho_w, ho_h);  // 外側 D
                    d2(hi_w, hi_h);  // 内側 D(指の入る空間)
                }
}
