```openscad
// マグカップのパラメータ
outer_d = 80;
inner_d = 70;
height = 90;
bottom_thickness = 6;

// 取手のパラメータ
handle_inner_h = 30; // 内側の高さ
handle_inner_w = 25; // 本体から外側への張り出し
handle_thick = 10;   // 取手の肉厚

module mug() {
    union() {
        // マグ本体
        difference() {
            cylinder(d = outer_d, h = height, $fn = 100);
            translate([0, 0, bottom_thickness])
                cylinder(d = inner_d, h = height - bottom_thickness + 0.1, $fn = 100);
        }

        // 取手
        // 本体外径（半径40mm）に接触させるため、X軸方向に配置
        // D字形状：本体側が直線、外側が半円
        translate([outer_d / 2 - 1, 0, height / 2]) {
            rotate([0, 90, 0]) {
                rotate([0, 0, 90]) {
                    linear_extrude(height = handle_thick) {
                        difference() {
                            // 取手の外形
                            hull() {
                                translate([-handle_inner_h / 2, 0]) circle(d = handle_thick, $fn = 50);
                                translate([handle_inner_h / 2, 0]) circle(d = handle_thick, $fn = 50);
                                translate([0, handle_inner_w + handle_thick / 2]) circle(d = handle_thick, $fn = 50);
                            }
                            // 取手の内側の穴
                            hull() {
                                translate([-handle_inner_h / 2 + (handle_thick/2), 0]) square([handle_inner_h - handle_thick, 0.1]);
                                translate([0, handle_inner_w]) circle(d = handle_inner_h - handle_thick, $fn = 50);
                            }
                        }
                    }
                }
            }
        }
    }
}

mug();
```
