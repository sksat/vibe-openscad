```openscad
$fn = 96;

leaf_length = 30;
leaf_width = 25;
leaf_thickness = 2;

knuckle_length = 6;
knuckle_outer_d = 8;
knuckle_inner_d = 4.6;

pin_d = 4;
pin_length = 32;

// 板の内側の縁を筒部に重ね、板と筒部を一体化する。
inner_edge = 3.5;

module leaf(side) {
    x_min = side < 0 ? -inner_edge - leaf_width : inner_edge;
    screw_x = side * (inner_edge + leaf_width - 6);

    difference() {
        union() {
            translate([x_min, 0, -leaf_thickness / 2])
                cube([leaf_width, leaf_length, leaf_thickness]);

            for (i = [0:4])
                if ((i % 2 == 0) == (side < 0))
                    translate([0, i * knuckle_length, 0])
                        rotate([-90, 0, 0])
                            cylinder(h = knuckle_length, d = knuckle_outer_d);
        }

        for (i = [0:4])
            if ((i % 2 == 0) == (side < 0))
                translate([0, i * knuckle_length - 0.01, 0])
                    rotate([-90, 0, 0])
                        cylinder(h = knuckle_length + 0.02,
                                 d = knuckle_inner_d);

        for (y = [7:8:23]) {
            translate([screw_x, y, -1.01])
                cylinder(h = 2.02, d = 3.2);

            // 上面から深さ 1 mm の皿穴
            translate([screw_x, y, 0])
                cylinder(h = 1.01, d1 = 3.2, d2 = 6);
        }
    }
}

color("silver") leaf(-1);
color("silver") leaf(1);

color("gray")
    translate([0, -1, 0])
        rotate([-90, 0, 0])
            cylinder(h = pin_length, d = pin_d);
```
