```openscad
$fn = 96;

eps = 0.01;

leaf_len   = 30;
leaf_w     = 25;
leaf_t     = 2;

pin_d      = 4;
pin_len    = 32;

knuckle_od = 8;
knuckle_id = 4.6;
knuckle_len = 6;

leaf_y0 = 1;          // leaves and knuckles span Y = 1..31
leaf_inner_x = 3;     // inner edge of flat leaf from hinge axis
hole_x = 21;          // countersunk hole center, far from knuckle
hole_ys = [8, 16, 24];

module y_cylinder(d, h) {
    rotate([90, 0, 0]) cylinder(d = d, h = h, center = true);
}

module knuckle_segment(idx) {
    yc = leaf_y0 + idx * knuckle_len + knuckle_len / 2;
    translate([0, yc, 0])
    difference() {
        y_cylinder(knuckle_od, knuckle_len);
        y_cylinder(knuckle_id, knuckle_len + 2 * eps);
    }
}

module mounting_hole() {
    union() {
        translate([0, 0, -leaf_t/2 - eps])
            cylinder(h = leaf_t + 2*eps, d = 3.2);
        translate([0, 0, -eps])
            cylinder(h = 1 + eps, d1 = 3.2, d2 = 6);
    }
}

module leaf_plate(side) {
    x0 = (side < 0) ? -(leaf_inner_x + leaf_w) : leaf_inner_x;
    translate([x0, leaf_y0, -leaf_t/2])
        cube([leaf_w, leaf_len, leaf_t]);
}

module leaf(side) {
    idxs = (side < 0) ? [0, 2, 4] : [1, 3];

    difference() {
        union() {
            leaf_plate(side);
            for (i = idxs)
                knuckle_segment(i);
        }

        for (yh = hole_ys)
            translate([side * hole_x, yh, 0])
                mounting_hole();
    }
}

module pin_axis() {
    translate([0, pin_len/2, 0])
        y_cylinder(pin_d, pin_len);
}

leaf(-1);
leaf( 1);
pin_axis();
```
