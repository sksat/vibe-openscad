```openscad
// ===============================================
// 小型バット蝶番 (butt hinge) - 開いた状態 180°
// 左板 / 右板 / ピン軸 の3パーツ構成
// ===============================================

$fn = 64;

// ---- 主要寸法 ----
leaf_len     = 30;    // 縦(ヒンジ軸方向)
leaf_width   = 25;    // 横(開く方向)
leaf_th      = 2;     // 板厚

pin_d        = 4;     // ピン直径
pin_len      = 32;    // ピン長(両端1mmずつ突出)

knuckle_od   = 8;     // knuckle 外径
clearance    = 0.3;
knuckle_id   = pin_d + clearance;   // 4.6

seg          = leaf_len / 5;        // 6mm
gap          = 0.15;                // 回転用の軸方向すきま

// ネジ穴(M3 皿穴)
screw_pitch  = 8;
screw_th_d   = 3.2;   // 貫通穴
cs_top_d     = 6;     // 皿穴上部径
cs_depth     = 1;     // テーパ深さ
screw_x_off  = leaf_width - 7;      // knuckle から離れた側の位置

// 板中心のZ: 板の平面が Z 軸に直交、ピン中心 Z=0 に対し板は上側に接する
leaf_z = knuckle_od/2 - leaf_th/2;  // 板の中心 Z

// ---- knuckle の Y 区間 ----
// 左板: セグメント 0, 2, 4   右板: セグメント 1, 3
function seg_y0(i) = -leaf_len/2 + i*seg;

// ===============================================
// モジュール
// ===============================================

// 皿穴(座ぐり) : 板の上面(+Z)から見て
module countersink() {
    union() {
        // 貫通穴
        translate([0, 0, -leaf_th])
            cylinder(d = screw_th_d, h = leaf_th*3);
        // テーパ部(上面から深さ cs_depth)
        translate([0, 0, leaf_th/2 - cs_depth])
            cylinder(d1 = screw_th_d, d2 = cs_top_d, h = cs_depth + 0.01);
        // 上側の逃げ
        translate([0, 0, leaf_th/2 - 0.01])
            cylinder(d = cs_top_d, h = 1);
    }
}

// 板本体(+X 方向に伸びる形。左板は mirror して使う)
module leaf_plate() {
    difference() {
        // 板: X = 0 .. leaf_width
        translate([0, -leaf_len/2, leaf_z - leaf_th/2])
            cube([leaf_width, leaf_len, leaf_th]);

        // 3 個の皿穴
        for (k = [-1, 0, 1])
            translate([screw_x_off, k * screw_pitch, leaf_z])
                countersink();
    }
}

// knuckle 群(指定セグメント番号のリスト)
module knuckles(idx) {
    for (i = idx)
        translate([0, seg_y0(i) + gap/2, 0])
            rotate([-90, 0, 0])
                cylinder(d = knuckle_od, h = seg - gap);
}

// knuckle と板をつなぐ部分(knuckle 上部を板厚まで埋める)
module knuckle_web(idx) {
    for (i = idx)
        translate([0, seg_y0(i) + gap/2, leaf_z - leaf_th/2])
            cube([knuckle_od/2, seg - gap, leaf_th]);
}

// 片側リーフ(板 + knuckle) : +X 方向版
module hinge_leaf(idx) {
    difference() {
        union() {
            leaf_plate();
            knuckles(idx);
            knuckle_web(idx);
        }
        // ピン穴(全長貫通)
        translate([0, -leaf_len/2 - 5, 0])
            rotate([-90, 0, 0])
                cylinder(d = knuckle_id, h = leaf_len + 10);
    }
}

// ---- ピン軸 ----
module hinge_pin() {
    translate([0, -pin_len/2, 0])
        rotate([-90, 0, 0])
            cylinder(d = pin_d, h = pin_len);
}

// ===============================================
// 組み立て(180° 開いた状態)
// ===============================================

// 右板 (x > 0) : 中間セグメント 1, 3
color("Silver")
    hinge_leaf([1, 3]);

// 左板 (x < 0) : 外側 0, 4 + 中央 2
color("LightSteelBlue")
    mirror([1, 0, 0])
        hinge_leaf([0, 2, 4]);

// ピン軸
color("DimGray")
    hinge_pin();
```
