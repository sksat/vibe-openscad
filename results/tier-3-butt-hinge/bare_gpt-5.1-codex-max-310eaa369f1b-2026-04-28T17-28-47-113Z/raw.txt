```openscad
$fn = 64;

module knuckle(ypos, length=6){
    difference(){
        translate([0, ypos, 1])
            rotate([90,0,0])
                cylinder(h=length, r=4, center=true);
        translate([0, ypos, 1])
            rotate([90,0,0])
                cylinder(h=length+0.4, r=2.3, center=true);
    }
}

module leaf(side = -1){ // side = -1 (left), +1 (right)
    // plate extents
    plate_origin = (side == -1) ? [-25, -15, 0] : [0, -15, 0];

    // knuckle positions
    left_idxs  = [0,2,4];
    right_idxs = [1,3];

    difference(){
        union(){
            // plate
            translate(plate_origin)
                cube([25, 30, 2]);

            // knuckles
            if(side == -1){
                for(i = left_idxs)
                    knuckle(-12 + 6*i);
            } else {
                for(i = right_idxs)
                    knuckle(-12 + 6*i);
            }
        }

        // countersunk holes
        for(ypos = [-8, 0, 8]){
            xpos = side * 15;
            // through hole
            translate([xpos, ypos, -1])
                cylinder(h=5, r=1.6, center=false);
            // countersink (from top surface, depth 1mm)
            translate([xpos, ypos, 1])
                cylinder(h=1, r1=3, r2=1.6, center=false);
        }
    }
}

// Assemble
leaf(-1);
leaf(1);

// Pin axis
translate([0,0,1])
    rotate([90,0,0])
        cylinder(h=32, r=2, center=true);
```
