```openscad
// =====================================================
// 家具用 小型バット蝶番 (Butt Hinge)
// 開いた状態 (180°) で出力
// =====================================================

// ---------- パラメータ ----------
leaf_width   = 25;    // 板の横(開く方向)
leaf_length  = 30;    // 板の縦(ピン軸方向)
leaf_thick   = 2;     // 板の厚さ

pin_dia      = 4;     // ピン軸 直径
pin_len      = 32;    // ピン軸 長さ(両端 1mm ずつ飛び出す)

knuckle_od   = 8;     // knuckle 外径
knuckle_clr  = 0.3;   // クリアランス
knuckle_id   = pin_dia + knuckle_clr;  // 4.6mm
knuckle_seg  = 6;     // 各 knuckle の長さ(30/5)

screw_csink_dia = 6;    // 皿穴 テーパ径
screw_csink_dep = 1;    // 皿穴 深さ
screw_thru_dia  = 3.2;  // 貫通穴径
screw_pitch     = 8;    // ねじ穴ピッチ

$fn = 64;

// knuckle のセグメント Y 位置(下端) : 0,6,12,18,24
// 互い違い配置
//   左板: セグメント 0, 2, 4 (外側2 + 中央1)
//   右板: セグメント 1, 3   (中間2)
left_segments  = [0, 2, 4];
right_segments = [1, 3];

// ---------- knuckle (筒部) ----------
module knuckle(seg_index) {
    y0 = seg_index * knuckle_seg;
    translate([0, y0, 0])
        rotate([-90, 0, 0])
            difference() {
                cylinder(h = knuckle_seg, d = knuckle_od);
                translate([0, 0, -1])
                    cylinder(h = knuckle_seg + 2, d = knuckle_id);
            }
}

// ---------- 皿穴 1個分 (Z方向に貫通、表面 z=+leaf_thick/2 側にテーパ) ----------
module screw_hole() {
    // 貫通穴
    translate([0, 0, -leaf_thick/2 - 1])
        cylinder(h = leaf_thick + 2, d = screw_thru_dia);
    // 皿(テーパ): 表面側
    translate([0, 0, leaf_thick/2 - screw_csink_dep])
        cylinder(h = screw_csink_dep + 0.01,
                 d1 = screw_thru_dia, d2 = screw_csink_dia);
    // 表面より上の余裕(テーパ口を確実に開ける)
    translate([0, 0, leaf_thick/2 - 0.001])
        cylinder(h = 0.5, d = screw_csink_dia);
}

// ---------- 板本体 (knuckle 側 x=0 から外側へ) ----------
// dir = -1 : 左板 (x<0)   dir = +1 : 右板 (x>0)
module leaf_plate(dir) {
    // knuckle 中心(X=0)から板が伸びる。
    // 板は knuckle 外周に接続するため、根元を knuckle_od/2 付近から始める。
    plate_start = knuckle_od/2 - 1;        // 少し knuckle に食い込ませて結合
    plate_x     = leaf_width;              // 板の伸び量

    difference() {
        // 板 (厚さは z 方向、上面 z=+leaf_thick/2)
        if (dir < 0) {
            translate([-plate_start - plate_x, 0, -leaf_thick/2])
                cube([plate_x, leaf_length, leaf_thick]);
        } else {
            translate([plate_start, 0, -leaf_thick/2])
                cube([plate_x, leaf_length, leaf_thick]);
        }

        // ねじ皿穴 3 個 (knuckle から離れた側)
        // X 位置: 板の外端寄り
        screw_x = dir * (plate_start + plate_x - 5);
        // Y 中心を中央、8mm ピッチで 3 個
        for (i = [-1, 0, 1]) {
            translate([screw_x, leaf_length/2 + i * screw_pitch, 0])
                screw_hole();
        }
    }
}

// ---------- 左板 アセンブリ ----------
module left_leaf() {
    color("Silver") {
        leaf_plate(-1);
        for (s = left_segments) knuckle(s);
    }
}

// ---------- 右板 アセンブリ ----------
module right_leaf() {
    color("LightSteelBlue") {
        leaf_plate(1);
        for (s = right_segments) knuckle(s);
    }
}

// ---------- ピン軸 ----------
module pin() {
    color("Gold")
        translate([0, -1, 0])      // -Y 側に 1mm 飛び出す
            rotate([-90, 0, 0])
                cylinder(h = pin_len, d = pin_dia);
}

// ---------- 組み立て (180° 開いた状態) ----------
left_leaf();
right_leaf();
pin();
```
