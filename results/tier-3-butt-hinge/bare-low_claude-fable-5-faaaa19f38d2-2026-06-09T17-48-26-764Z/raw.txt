```openscad
// 小型バット蝶番(開いた状態 180°)
$fn = 64;

leaf_w   = 25;    // 横幅
leaf_l   = 30;    // 縦(Y方向)
leaf_t   = 2;     // 板厚
pin_d    = 4;
pin_l    = 32;
kn_od    = 8;     // knuckle 外径
kn_id    = 4.6;   // knuckle 内径(クリアランス込み)
seg      = 6;     // knuckle 1個の長さ
hole_x   = 20;    // ネジ穴の軸からの距離
hole_ys  = [7, 15, 23];

// knuckle 1個(Y=y0 から長さ seg)
module knuckle(y0) {
    translate([0, y0, 0])
        rotate([-90, 0, 0])
            difference() {
                cylinder(h = seg, d = kn_od);
                translate([0, 0, -0.1])
                    cylinder(h = seg + 0.2, d = kn_id);
            }
}

// 皿穴付きネジ穴
module screw_hole(x, y) {
    translate([x, y, 0]) {
        translate([0, 0, -leaf_t])
            cylinder(h = leaf_t * 2, d = 3.2);          // 貫通穴
        translate([0, 0, leaf_t/2 - 1])
            cylinder(h = 1 + 0.1, d1 = 4, d2 = 6);      // テーパ皿(表面 d6)
    }
}

// 板 + knuckle(side = -1: 左, +1: 右)
module leaf(side, kn_ys) {
    difference() {
        union() {
            // 平板
            translate([side > 0 ? 0 : -leaf_w, 0, -leaf_t/2])
                cube([leaf_w, leaf_l, leaf_t]);
            // knuckle
            for (y = kn_ys) knuckle(y);
        }
        // ピン穴(knuckle 全長を通す)
        translate([0, -1, 0])
            rotate([-90, 0, 0])
                cylinder(h = pin_l + 2, d = kn_id);
        // 反対側の knuckle と干渉しないよう板側もピン軸部をクリア
        // ネジ穴
        for (y = hole_ys)
            screw_hole(side * hole_x, y);
    }
}

// 左板: knuckle 3個(両端 + 中央)
leaf(-1, [0, 12, 24]);

// 右板: knuckle 2個(中間)
leaf(1, [6, 18]);

// ピン軸
translate([0, -1, 0])
    rotate([-90, 0, 0])
        cylinder(h = pin_l, d = pin_d);
```
