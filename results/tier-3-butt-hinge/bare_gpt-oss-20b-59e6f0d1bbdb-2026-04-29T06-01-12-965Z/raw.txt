```openscad
// BUTT HINGE for furniture (180° open state)

module pin_axis() {
    // 4mm diameter, length 32mm with 1mm protrusion on each side
    translate([0,-16,0]) cylinder(h=32, r=2, $fn=64);
}

module knuckle(x_offset) {
    // outer radius 4mm (diameter 8mm), inner hole 4.6mm diameter
    difference() {
        translate([x_offset,0,0])
            cylinder(h=30, r=4, $fn=64); // outer cylinder
        translate([x_offset,0,-1]) // slightly longer to ensure full cut
            cylinder(h=32, r=2.3, $fn=64); // inner hole (4.6mm dia)
    }
}

module leaf(side="left") {
    // side: "left" or "right"
    // leaf dimensions 30x25x2 mm
    color(side=="left"?["red","gray"]:"blue")
    translate([side=="left"?-12.5:12.5, -15, -1]) // center at z=-1 for thickness 2mm
        cube([25,30,2], center=false);
}

module m3_hole(position) {
    // position: [x,y]
    translate([position[0], position[1], -0.5])
        union() {
            // tap: 6mm dia, 1mm deep taper (simplified as cylinder)
            cylinder(h=1, r=3, $fn=32);
            // through hole
            translate([0,0,-10]) cylinder(h=20, r=1.6, $fn=32);
        }
}

module m3_holes(side="left") {
    y_start = -8; // first hole at y=-8 from center
    for (i=[0:2]) {
        y_pos = y_start + i*8;
        x_pos = side=="left" ? -12.5 : 12.5;
        m3_hole([x_pos, y_pos]);
    }
}

// Main assembly
module butt_hinge() {
    // pin axis
    pin_axis();

    // knuckles (positions along Y at multiples of 6mm)
    for (i=[0:4]) {
        knuckle(i*6 - 12); // positions: -12, -6, 0, 6, 12
    }

    // leaves
    leaf("left");
    leaf("right");

    // M3 holes on each side
    m3_holes("left");
    m3_holes("right");
}

// Render the hinge in 180° open state
butt_hinge();
```
