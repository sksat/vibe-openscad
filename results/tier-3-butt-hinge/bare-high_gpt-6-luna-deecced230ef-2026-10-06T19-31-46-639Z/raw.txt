```openscad
$fn = 64;

// Dimensions (mm)
leaf_length = 30;
leaf_width = 25;
leaf_thickness = 2;

pin_diameter = 4;
pin_length = 32;

knuckle_diameter = 8;
knuckle_bore_diameter = 4.6;
knuckle_length = 6;

leaf_inner_edge = knuckle_diameter / 2 + 0.15;
leaf_outer_edge = leaf_inner_edge + leaf_width;

module y_cylinder(length, diameter) {
    rotate([90, 0, 0])
        cylinder(h = length, d = diameter, center = true);
}

module leaf_plate(side) {
    hole_x = side * (leaf_outer_edge - 4);

    difference() {
        if (side < 0)
            translate([-leaf_outer_edge, -leaf_length / 2, -leaf_thickness / 2])
                cube([leaf_width, leaf_length, leaf_thickness]);
        else
            translate([leaf_inner_edge, -leaf_length / 2, -leaf_thickness / 2])
                cube([leaf_width, leaf_length, leaf_thickness]);

        for (hole_y = [-8, 0, 8]) {
            translate([hole_x, hole_y, -leaf_thickness / 2 - 0.01])
                cylinder(h = leaf_thickness + 0.02, d = 3.2);

            translate([hole_x, hole_y, 0])
                cylinder(h = 1, d1 = 3.2, d2 = 6);
        }
    }
}

module leaf_knuckles(side) {
    for (i = [0:4]) {
        if ((side < 0 && (i == 0 || i == 2 || i == 4)) ||
            (side > 0 && (i == 1 || i == 3))) {
            segment_y = -leaf_length / 2 + i * knuckle_length;
            segment_center_y = segment_y + knuckle_length / 2;

            difference() {
                translate([0, segment_center_y, 0])
                    y_cylinder(knuckle_length, knuckle_diameter);

                translate([0, segment_center_y, 0])
                    y_cylinder(knuckle_length + 0.02, knuckle_bore_diameter);
            }

            // Attachment web, present only on this leaf's knuckle segments.
            if (side < 0)
                translate([-4.25, segment_y, -leaf_thickness / 2])
                    cube([0.85, knuckle_length, leaf_thickness]);
            else
                translate([3.4, segment_y, -leaf_thickness / 2])
                    cube([0.85, knuckle_length, leaf_thickness]);
        }
    }
}

module hinge_leaf(side) {
    union() {
        leaf_plate(side);
        leaf_knuckles(side);
    }
}

color("silver")
    hinge_leaf(-1);

color("gainsboro")
    hinge_leaf(1);

color("dimgray")
    y_cylinder(pin_length, pin_diameter);
```
