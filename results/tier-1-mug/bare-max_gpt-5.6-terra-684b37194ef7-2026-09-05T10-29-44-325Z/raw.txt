```openscad
$fn = 128;

outer_diameter = 80;
cup_height = 100;
wall_thickness = 4;
bottom_thickness = 6;

outer_radius = outer_diameter / 2;
inner_radius = outer_radius - wall_thickness;

handle_ring_radius = 12;
handle_tube_radius = 2;
handle_center_x = outer_radius + handle_ring_radius;

module cup_body() {
    difference() {
        cylinder(r = outer_radius, h = cup_height);

        translate([0, 0, bottom_thickness])
            cylinder(
                r = inner_radius,
                h = cup_height - bottom_thickness + 1
            );
    }
}

module ring_handle() {
    translate([handle_center_x, 0, cup_height / 2])
        rotate([90, 0, 0])
            rotate_extrude(convexity = 10)
                translate([handle_ring_radius, 0, 0])
                    circle(r = handle_tube_radius);
}

union() {
    cup_body();
    ring_handle();
}
```
