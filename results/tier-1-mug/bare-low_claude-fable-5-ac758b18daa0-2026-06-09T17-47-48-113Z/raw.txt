```openscad
// マグカップ
// 外径 80mm (半径 40mm)、高さ 100mm、肉厚 4mm、底厚 6mm
// 取手: リング半径 12mm のトーラス状

$fn = 100;

cup_outer_r   = 40;   // 外半径
cup_h         = 100;  // 高さ
wall_t        = 4;    // 側面肉厚
bottom_t      = 6;    // 底面厚み

handle_r      = 12;   // 取手リングの半径(中心円の半径)
handle_thick  = 5;    // 取手の太さ(断面半径)

union() {
    // カップ本体
    difference() {
        cylinder(h = cup_h, r = cup_outer_r);
        // 内側くり抜き
        translate([0, 0, bottom_t])
            cylinder(h = cup_h - bottom_t + 1, r = cup_outer_r - wall_t);
    }

    // 取手(トーラス)
    // カップ中央高さに配置し、外周から外側にはみ出させる
    translate([cup_outer_r, 0, cup_h / 2])
        rotate([90, 0, 0])
            rotate_extrude(angle = 360)
                translate([handle_r, 0, 0])
                    circle(r = handle_thick);
}
```
