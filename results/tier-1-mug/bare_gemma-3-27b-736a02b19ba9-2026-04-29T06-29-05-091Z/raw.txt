```openscad
// マグカップのパラメータ
outer_diameter = 80;
height = 100;
wall_thickness = 4;
bottom_thickness = 6;
handle_radius = 12;

// 内径
inner_diameter = outer_diameter - 2 * wall_thickness;

// モジュール：マグカップ本体
module mug() {
  difference() {
    union() {
      cylinder(h = height, r = outer_diameter / 2);
      cylinder(h = bottom_thickness, r = outer_diameter / 2);
    }
    cylinder(h = height + bottom_thickness, r = inner_diameter / 2);
  }
}

// モジュール：取手
module handle() {
  translate([0, 0, height/2])
  rotate_extrude()
  circle(r=handle_radius);
}


// メイン処理
union(){
    mug();
    handle();
}
```
