```openscad
$fn = 120;

// パラメータ
outer_d  = 80;    // カップ外径
cup_h    = 100;   // カップ高さ
wall     = 4;     // 側面の肉厚
bottom_t = 6;     // 底面の厚み

handle_r = 12;    // 取手リングの半径
handle_t = 5;     // 取手の太さ(管の半径)

// カップ本体(内側をくり抜き)
module cup_body() {
    difference() {
        // 外形
        cylinder(d = outer_d, h = cup_h);
        // 内側のくり抜き(底面の厚みを残す)
        translate([0, 0, bottom_t])
            cylinder(d = outer_d - 2 * wall, h = cup_h);
    }
}

// リング状の取手(トーラス)
module handle() {
    translate([outer_d / 2 + 6, 0, cup_h / 2])
        rotate([90, 0, 0])
            rotate_extrude()
                translate([handle_r, 0])
                    circle(r = handle_t);
}

// 組み立て
union() {
    cup_body();
    // 取手がカップ内部に飛び出さないように、内側空間と重なる部分を除去
    difference() {
        handle();
        translate([0, 0, bottom_t])
            cylinder(d = outer_d - 2 * wall, h = cup_h);
    }
}
```
