```openscad
$fn = 128;

outer_diameter = 80;
cup_height = 100;
wall_thickness = 4;
bottom_thickness = 6;

handle_outer_radius = 12;
handle_inner_radius = 9;
handle_thickness = 8;
handle_center_x = 50;

union() {
    // 上部を開け、底に6mmの厚みを残す
    difference() {
        cylinder(d = outer_diameter, h = cup_height);

        translate([0, 0, bottom_thickness])
            cylinder(
                d = outer_diameter - 2 * wall_thickness,
                h = cup_height - bottom_thickness + 1
            );
    }

    // 側面に接続するリング状の取手
    translate([handle_center_x, 0, cup_height / 2])
        rotate([90, 0, 0])
            linear_extrude(height = handle_thickness, center = true)
                difference() {
                    circle(r = handle_outer_radius);
                    circle(r = handle_inner_radius);
                }
}
```
