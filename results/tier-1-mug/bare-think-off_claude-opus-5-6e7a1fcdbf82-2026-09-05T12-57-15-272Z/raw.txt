```openscad
// ===== マグカップ パラメータ =====
outer_d      = 80;   // 外径
outer_r      = outer_d / 2;
height       = 100;  // 全高
wall         = 4;    // 側面肉厚
bottom       = 6;    // 底面厚み

inner_r      = outer_r - wall;      // 内径半径 = 36
inner_h      = height - bottom;     // 内側空間の高さ = 94

// ===== 取手パラメータ =====
handle_r       = 12;   // 取手リングの半径(中心線の円半径)
handle_thick   = 9;    // 取手の太さ(断面直径)
handle_z       = height / 2;              // 取手の高さ位置(カップ中心高さ)
handle_offset  = outer_r + handle_r - 5;  // 取手リング中心のX位置(外周から十分外へ)

$fn = 120;

// ===== カップ本体 =====
module cup_body() {
    difference() {
        cylinder(h = height, r = outer_r);
        translate([0, 0, bottom])
            cylinder(h = inner_h + 1, r = inner_r);
    }
}

// ===== 取手(リング状) =====
module handle_ring() {
    translate([handle_offset, 0, handle_z])
        rotate([90, 0, 0])
            rotate_extrude(convexity = 10)
                translate([handle_r, 0, 0])
                    circle(r = handle_thick / 2);
}

// ===== 組み立て =====
module mug() {
    union() {
        cup_body();
        // 取手のうちカップ内部に入り込む部分を削除
        difference() {
            handle_ring();
            translate([0, 0, -1])
                cylinder(h = height + 2, r = inner_r);
        }
    }
}

mug();
```
