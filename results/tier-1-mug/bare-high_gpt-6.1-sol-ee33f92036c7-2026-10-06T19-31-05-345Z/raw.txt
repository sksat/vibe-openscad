```openscad
$fn = 128;

outer_diameter = 80;
cup_height = 100;
wall_thickness = 4;
bottom_thickness = 6;

handle_radius = 12;
handle_tube_radius = 3;

outer_radius = outer_diameter / 2;
inner_radius = outer_radius - wall_thickness;
epsilon = 0.01;

union() {
    difference() {
        cylinder(
            h = cup_height,
            r = outer_radius
        );

        translate([0, 0, bottom_thickness])
            cylinder(
                h = cup_height - bottom_thickness + epsilon,
                r = inner_radius
            );
    }

    translate([
        outer_radius + handle_radius,
        0,
        cup_height / 2
    ])
        rotate([90, 0, 0])
            rotate_extrude(convexity = 10)
                translate([handle_radius, 0, 0])
                    circle(r = handle_tube_radius);
}
```
