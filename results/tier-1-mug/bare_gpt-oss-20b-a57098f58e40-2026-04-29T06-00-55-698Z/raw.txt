```openscad
$fn = 60;   // 精度設定

// 基本パラメータ
outer_diameter = 80;
wall_thickness  = 4;
bottom_thick    = 6;
height          = 100;

// 半径計算
outer_r = outer_diameter / 2;
inner_r = outer_r - wall_thickness;

module cup() {
    difference(){
        // 外側のカップ本体
        cylinder(h=height, r=outer_r, center=false);
        // 内側を抜き、底面厚みを保持
        translate([0,0,bottom_thick])
            cylinder(h=height - bottom_thick, r=inner_r, center=false);
    }
}

module handle() {
    // 取手（リング状）を中心高さに付ける
    translate([0,0,height/2])
        rotate_extrude(angle=360,$fn=120){
            translate([outer_r + 12, 0])   // 外径方向に12mm伸びる位置へ移動
                circle(r=4);               // 取手の半径（例：4mm）
        }
}

union(){
    cup();
    handle();
}
```
