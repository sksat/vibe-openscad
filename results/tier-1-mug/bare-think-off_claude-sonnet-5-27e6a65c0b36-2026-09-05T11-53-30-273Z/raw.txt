```openscad
// マグカップのパラメータ
outer_diameter = 80;
outer_radius = outer_diameter / 2;
height = 100;
wall_thickness = 4;
bottom_thickness = 6;

inner_radius = outer_radius - wall_thickness;
inner_height = height - bottom_thickness;

// 取手のパラメータ
handle_tube_radius = 12; // 取手の管の半径(太さ)
handle_ring_radius = 28; // 取手のリング自体の半径(大きさ)
handle_center_height = height / 2;

module mug_body() {
    difference() {
        // 外側の円柱
        cylinder(r=outer_radius, h=height, $fn=100);
        
        // 内側をくり抜く(底面の厚みを残す)
        translate([0, 0, bottom_thickness])
            cylinder(r=inner_radius, h=inner_height + 1, $fn=100);
    }
}

module handle() {
    translate([outer_radius + handle_ring_radius - handle_tube_radius, 0, handle_center_height])
        rotate([90, 0, 0])
            rotate_extrude($fn=100)
                translate([handle_ring_radius, 0, 0])
                    circle(r=handle_tube_radius, $fn=50);
}

module mug() {
    union() {
        mug_body();
        handle();
    }
}

mug();
```
