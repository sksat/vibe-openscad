```openscad
// ============================================
//  マグカップ (外径80 / 高さ100 / 肉厚4 / 底6)
// ============================================

$fn = 96;

// ---- 基本寸法 ----
outer_d   = 80;              // 外径
outer_r   = outer_d / 2;     // 外半径 = 40
height    = 100;             // 全高
wall      = 4;               // 側面の肉厚
bottom_t  = 6;               // 底の厚み
inner_r   = outer_r - wall;  // 内半径 = 36

// ---- 取手(リング状)----
handle_ring_r = 12;                    // リング中心線の半径
handle_tube_r = 5;                     // 取手の太さ(断面半径)
handle_z      = height / 2;            // カップ中心高さに取り付け
handle_x      = outer_r + 9;           // 外周から外側へオフセット
                                       // → 指穴(半径 handle_ring_r - handle_tube_r)が
                                       //   カップ外周より外側に出る

// ---- モジュール ----
module cup_body() {
    cylinder(h = height, r = outer_r);
}

module cavity() {
    translate([0, 0, bottom_t])
        cylinder(h = height - bottom_t + 1, r = inner_r);
}

module handle() {
    translate([handle_x, 0, handle_z])
        rotate([90, 0, 0])
            rotate_extrude(convexity = 8)
                translate([handle_ring_r, 0])
                    circle(r = handle_tube_r);
}

// ---- 組み立て ----
difference() {
    union() {
        cup_body();
        handle();          // 取手は壁に食い込ませて確実に接合
    }
    cavity();              // 内側をくり抜く(取手の食い込み分も同時に整形)
}
```
